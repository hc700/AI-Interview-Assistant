/**
* Definition for singly-linked list.
* public class ListNode { int val; ListNode next; }
*/
class Solution {
    public ListNode reverseList(ListNode head) {
        ListNode prev = null; // 前驱指针
        ListNode cur = head; // 当前指针
        while (cur != null) {
            ListNode nxt = cur.next; // 暂存后继
            cur.next = prev; // 反转指针
            prev = cur;
            cur = nxt;
        }
        return prev;
    }
}
